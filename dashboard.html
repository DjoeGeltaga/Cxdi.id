async function saveConfig(e, type) { e.preventDefault(); try { if (type === 'links') { const { error: err1 } = await supabase.from('site_config').upsert({ key: 'wa', value: document.getElementById('waLink').value }, { onConflict: 'key' }); const { error: err2 } = await supabase.from('site_config').upsert({ key: 'discord', value: document.getElementById('discordLink').value }, { onConflict: 'key' }); if (err1 || err2) throw new Error((err1?.message || '') + ' ' + (err2?.message || '')); alert("Link berhasil disimpan secara online!"); loadData(); return; } let key = type; let value = ''; if (type === 'event') { const evData = { img: document.getElementById('evImg').value, name: document.getElementById('evName').value, date: document.getElementById('evDate').value, cls: document.getElementById('evClass').value, prize: document.getElementById('evPrize').value, rules: document.getElementById('evRules').value, reg: document.getElementById('evReg').value }; value = JSON.stringify(evData); } else if (type === 'schedule') { value = document.getElementById('schedDesc').value; } else if (type === 'rules') { value = document.getElementById('groupRules').value; } else if (type === 'sponsors') { value = document.getElementById('sponsorList').value; } const { error } = await supabase.from('site_config').upsert({ key: key, value: value }, { onConflict: 'key' }); if (error) { alert("Gagal Supabase: " + error.message); } else { alert("Data berhasil dipublikasikan secara global!"); loadData(); } } catch (err) { alert("Error tertangkap: " + err.message); } }
